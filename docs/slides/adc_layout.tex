\newcommand{\layoutframe}[5]{%
  \begin{frame}
    \frametitle{#1}
    \centering
    \begin{tikzpicture}
      \node[inner sep=0pt] (layout) {\includegraphics[height=#5\textheight,keepaspectratio]{#2}};
      \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]},line width=0.7pt,draw=NordBlue]
        ([yshift=-0.22cm]layout.south west) --
        node[below,font=\small,text=NordBody]{#3}
        ([yshift=-0.22cm]layout.south east);
      \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]},line width=0.7pt,draw=NordBlue]
        ([xshift=-0.22cm]layout.south west) --
        node[left,font=\small,text=NordBody]{#4}
        ([xshift=-0.22cm]layout.north west);
    \end{tikzpicture}
  \end{frame}%
}
