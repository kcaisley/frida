\documentclass[tikz,border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{xcolor}
\usetikzlibrary{arrows.meta}

\definecolor{NordBody}{HTML}{4C566A}
\definecolor{NordBlue}{HTML}{5E81AC}
\definecolor{NordPanel}{HTML}{ECEFF4}

\begin{document}
\begin{tikzpicture}[
  x=0.105cm,
  y=0.72cm,
  font=\sffamily\scriptsize,
  waveform/.style={draw=NordBody, line width=0.9pt, line join=miter, line cap=rect},
  guide/.style={draw=NordPanel, line width=0.35pt},
  major guide/.style={draw=NordBody!22, line width=0.45pt},
]
  % Each sequencer window is 2.5 ns. LOGIC is shifted by the propagation
  % delay t_pd while retaining a 2.5 ns high time and a 5 ns period.
  \def\logicdelay{1.25}
  \draw[guide] (-2.5,-0.30) -- (-2.5,3.68);
  \foreach \t in {0,2.5,...,102.5} {
    \draw[guide] (\t,-0.30) -- (\t,3.68);
  }
  \foreach \t in {0,10,...,100} {
    \draw[major guide] (\t,-0.30) -- (\t,3.68);
    \draw[NordBody, line width=0.45pt] (\t,-0.38) -- (\t,-0.28);
    \node[anchor=north, text=NordBody] at (\t,-0.41) {\t};
  }
  \draw[NordBody, line width=0.45pt] (102.5,-0.38) -- (102.5,-0.28);
  \node[anchor=north east, text=NordBody] at (102.5,-0.92) {Time (ns)};

  \foreach \y/\name in {3/INIT,2/SAMP,1/COMP,0/LOGIC} {
    \draw[guide] (-2.5,\y) -- (102.5,\y);
    \draw[guide] (-2.5,{\y+0.55}) -- (102.5,{\y+0.55});
    \node[anchor=east, font=\ttfamily\bfseries\scriptsize, text=NordBlue]
      at (-5,{\y+0.27}) {\name};
  }

  % INIT is high for one 2.5 ns window at the start of each sequence.
  \draw[waveform]
    (-2.5,3) -- (0,3) -- (0,3.55) -- (2.5,3.55) -- (2.5,3) --
    (100,3) -- (100,3.55) -- (102.5,3.55);

  % SAMP remains high for six windows, providing a 15 ns acquisition time.
  \draw[waveform]
    (-2.5,2) -- (2.5,2) -- (2.5,2.55) -- (17.5,2.55) -- (17.5,2) -- (102.5,2);
  \draw[{Stealth[length=1.2mm]}-{Stealth[length=1.2mm]}, draw=NordBlue, line width=0.55pt]
    (3.25,2.30) -- node[below, text=NordBlue, font=\sffamily\tiny]{15 ns} (16.75,2.30);

  % Seventeen COMP windows alternate with sixteen conversion LOGIC windows.
  % The initialization LOGIC pulse at 0--2.5 ns is separate.
  \draw[waveform] (-2.5,1) -- (17.5,1);
  \foreach \k in {0,...,16} {
    \pgfmathsetmacro{\compstart}{17.5 + 5*\k}
    \pgfmathsetmacro{\compstop}{20 + 5*\k}
    \pgfmathsetmacro{\pairstop}{22.5 + 5*\k}
    \draw[waveform]
      (\compstart,1) -- (\compstart,1.55) -- (\compstop,1.55) -- (\compstop,1) -- (\pairstop,1);
  }
  \draw[waveform]
    (-2.5,0) -- (0,0) -- (0,0.55) -- (2.5,0.55) -- (2.5,0) -- ({20+\logicdelay},0);
  \fill[NordBlue!14]
    (20,0.03) rectangle ({20+\logicdelay},0.52);
  \foreach \k in {0,...,15} {
    \pgfmathsetmacro{\logicstart}{20 + \logicdelay + 5*\k}
    \pgfmathsetmacro{\logicstop}{22.5 + \logicdelay + 5*\k}
    \ifnum\k<15
      \pgfmathsetmacro{\nextlogicstart}{25 + \logicdelay + 5*\k}
      \draw[waveform]
        (\logicstart,0) -- (\logicstart,0.55) -- (\logicstop,0.55) -- (\logicstop,0) -- (\nextlogicstart,0);
    \else
      \draw[waveform]
        (\logicstart,0) -- (\logicstart,0.55) -- (\logicstop,0.55) -- (\logicstop,0) -- (100,0);
    \fi
  }
  \draw[waveform] (100,0) -- (100,0.55) -- (102.5,0.55);

  \draw[-{Stealth[length=1.2mm]}, draw=NordBlue, line width=0.45pt]
    (30,0.76) node[anchor=west, text=NordBlue, font=\sffamily\tiny]{LOGIC edges offset by 1.25 ns}
    -- ({20+0.5*\logicdelay},0.28);
  \node[anchor=south, text=NordBody, font=\sffamily\tiny] at (101.25,3.70) {next sequence};
\end{tikzpicture}
\end{document}
