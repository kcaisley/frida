% The StrongARM latch as flow/comp/subckt.py builds it (NMOS input, switched tail; FRIDA-1's): the
% input pair on a clocked tail (with the tail's precharge PMOS), PMOS resets on its drains P and Q
% and on the latch nodes X and Y, and the cross-coupled NMOS (latch_on) and PMOS (latch_init).
% Short labels name the sizing parameters: in (diffpair), tail, rst, on (latch_on), init
% (latch_init).
\input{drawing-preamble}
\usepackage{thesis-circuits}
\begin{document}
\begin{drawing}
  \begin{tikzpicture}[circuit]
    \comppreamp
    % the latch: cross-coupled NMOS over P and Q, cross-coupled PMOS to the supply
    \draw (-1.6,3.51) node[nmos, xscale=-1](m3){};
    \draw (1.6,3.51) node[nmos](m4){};
    \draw (m3.S) -- (P) (m4.S) -- (Q);
    \draw (-1.6,6.25) node[pmos, xscale=-1](m5){};
    \draw (1.6,6.25) node[pmos](m6){};
    \draw (m5.S) node[vcc]{} (m6.S) node[vcc]{};
    \draw (m3.D) -- (m5.D) (m4.D) -- (m6.D);
    % gate ties, then X to the right tie and Y to the left tie, each jumping the other tie
    \draw (m3.G) -- (m5.G) (m4.G) -- (m6.G);
    \coordinate (X) at ($(m3.D)!0.75!(m5.D)$);
    \coordinate (Y) at ($(m4.D)!0.45!(m6.D)$);
    \draw (X) -- (X -| m3.G) node[jump crossing]{} -- (X -| m4.G);
    \draw (Y) -- (Y -| m4.G) node[jump crossing]{} -- (Y -| m3.G);
    \fill (X) circle (1.6pt) (Y) circle (1.6pt) (X -| m4.G) circle (1.6pt) (Y -| m3.G) circle (1.6pt);
    % X and Y, with the reset PMOS that precharge them
    \draw (-4.2,5.95) node[pmos](mrx){};
    \draw (mrx.D) |- (X);
    \draw (mrx.S) node[vcc]{};
    \draw (mrx.G) node[left, sig]{CLK};
    \draw (4.2,5.6) node[pmos, xscale=-1](mry){};
    \draw (mry.D) |- (Y);
    \draw (mry.S) node[vcc]{};
    \draw (mry.G) node[right, sig]{CLK};
    \draw (X) node[above left, sig]{$X$};
    \draw (Y) node[below right, sig]{$Y$};
    % the sizing parameters, beside their devices
    \node[left, note] at ($(m3.S)!0.5!(m3.D) - (0.12,0)$) {on};
    \node[left, note] at ($(m5.S)!0.5!(m5.D) - (0.12,0)$) {init};
    \foreach \m/\t in {mrx/rst, mry/rst} {\node[below, note] at ($(\m.G) - (0,0.1)$) {\t};}
  \end{tikzpicture}
\end{drawing}
\end{document}
