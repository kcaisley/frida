\usepackage{thesis-circuits}
\begin{document}
\begin{drawing}
  \begin{tikzpicture}[circuit]
    % the switch: source at the input, drain at the held node, gate on the sampling clock
    \draw (1.6,0) node[nmos, rotate=-90](mn){};
    \draw (0,0) node[left, sig]{$v_\mathrm{in}$} to[short, o-] (mn.S);
    \draw (mn.G) node[above, sig]{$\phi$};

    \begin{scope}[densely dashed]
      \draw (4.6,1.0) node[nmos, rotate=-90](md){};
      \draw (md.S) -- (md.S |- out);
      \draw (md.D) -- (md.D |- out);
    \end{scope}
    \fill (md.S |- out) circle (1.6pt) (md.D |- out) circle (1.6pt);
    \draw (md.G) node[above, sig]{$\overline{\phi}$};
    \draw (md.D) ++(0.15,0.25) node[right, note]{dummy\\$W/2$};
