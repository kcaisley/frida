% A comparator's first stage with the dynamic bias (Bindra; flow/comp/subckt.py, Bias.DYNAMIC): the
% clocked tail's source on a node S held low during reset (an NMOS on the inverted clock, the tail's
% size) and loaded by C_TAIL to the supply, which stops the input pair once it has shared its charge;
% C_P loads each drain (P, Q) to ground. Both are MOS capacitors (bias_c, preamp_c). There is no
% tail precharge in this variant.

    \draw (mt.G) node[left, sig]{CLK};
    \draw (mb.D) -- (mt.S);
    \coordinate (S) at ($(mb.D)!0.5!(mt.S)$);
    \draw (S) -- (2.4,0.885) to[C, l_=$C_\mathrm{TAIL}$] (2.4,2.6) node[vcc]{};
    \fill (S) circle (1.6pt);
    \draw (S) node[left, sig]{$S$};
    \draw (mt.D) -- (0,2.9) coordinate(tail);
